Write SHORT, SPECIFIC, HIGH-IMPACT messages in EXACTLY 3 SENTENCES OR FEWER.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   ✅ "2,100-patient trial showed 38% caries reduction — JIDA Oct 2026 p.14"

   Rule: If you have a number, use it. If you have a source name, cite it.

   Rule: Use ONLY numbers that exist in the VERIFIED FACTS. Never invent a number

③ MERCHANT FIT (0-10): Personalise to THIS exact merchant.
   - Use owner_first_name (e.g. "Dr. Meera,", "Lakshmi,", "Karthik,").
   - Reference THEIR numbers: their CTR, their lapsed count, their active offer.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# ─────────────────────────────────────────────────────────────────────────────
# MODE INSTRUCTIONS — injected based on intent state from /reply routing
# ─────────────────────────────────────────────────────────────────────────────

_MODE = {
    "normal": (
        "Compose the initial outbound message. "
        "Lead with the trigger fact. Close with a single binary CTA."
    ),
    "action": (
        "MERCHANT COMMITTED (said yes/confirmed/go ahead). "
        "Do NOT ask more qualifying questions. "
        "Tell them what you are doing RIGHT NOW. "
        "Switch to execution language: 'I've drafted...', 'Setting up...', 'Done — here's...'."
    ),
    "answer": (
        "Merchant asked a question. "
        "Answer it directly and concisely in the first sentence. "
        "Then re-offer the CTA in the last sentence."
    ),
    "soft_exit": (
        "Merchant expressed reluctance or objection. "
        "Acknowledge it warmly in one sentence. "
        "Leave the door open without hard-selling. "
        "Close gracefully: 'Happy to revisit when the time is right.'"
    ),
}

_USER_TEMPLATE = """VERIFIED FACTS (ONLY use data from here — everything else is fabrication):
{facts}

PRIOR MESSAGES IN THIS CONVERSATION (do NOT repeat or rephrase these):
{prior_bodies}

ROUTER REASON (why the router said PROCEED): {router_reason}
INTENT CONTEXT: {intent_context}
COMPOSITION MODE: {mode_instruction}

FRAMING HINT: {framing_hint}
Write the WhatsApp message now. Max 3 sentences. No URLs. Use the owner's name."""


# ─────────────────────────────────────────────────────────────────────────────
# COMPOSE FUNCTION
# ─────────────────────────────────────────────────────────────────────────────

async def compose(
    facts: dict,
    conversation_id: str,
    merchant_id: str,
    trigger_id: str,
    customer_id: Optional[str] = None,
    router_reason: str = "",
    intent_context: str = "",
    mode: str = "normal",
) -> Optional[ComposedMessage]:
    """
    Compose a WhatsApp message from verified facts.

    Returns ComposedMessage on success.
    Returns None on timeout or if all retries fail (instructor exhausts max_retries=1).

    The asyncio.wait_for(timeout=12.0) budget is applied by the caller (main.py),
    not here — this keeps the budget shared between router + composer.
    """

    # Build anti-repetition context (last 3 Vera turns)
    history = state.get_conversation_history(conversation_id)
    prior_vera = [t["body"] for t in history if t.get("role") == "vera"]
    prior_str = (
        "\n".join(f"  - {b[:160]}" for b in prior_vera[-3:])
        if prior_vera
        else "  (none — this is the first message)"
    )

    # Inject trigger-type-aware framing hints directly into the user message.
    # System prompt prohibitions alone are insufficient — a positive example
    # in the user turn overrides gpt-4o-mini's training patterns.
    trigger_kind = facts.get("trigger_kind", "")
    has_customer  = bool(facts.get("customer_id") or customer_id)

    if has_customer:
        # Customer-scope: prevent merchant/customer persona confusion
        framing_hint = (
            "customer_id IS present — write TO the merchant ABOUT their customer. "
            "Sentence 1 example structure: '[Owner], your [member/patient/customer] "
            "[name if available] [payload fact] din se nahi aya/hasn't visited.' "
            "Sentence 3 MUST be a binary YES/NO, NOT an open-ended question: "
            "'I have drafted a [win-back/recall] message — reply YES to send it or NO to skip.'"
        )
    elif trigger_kind in ("competitor_opened", "perf_dip", "review_theme_emerged"):
        # Consequence triggers: prevent hedge language in sentence 2
        framing_hint = (
            "Sentence 2 must state what IS already happening, not what 'could' happen. "
            "Do NOT write 'could impact', 'might affect', 'may influence'. "
            "Use present tense: 'patients searching for X will see their listing first' "
            "or 'this puts your [metric] at risk this week'."
        )
    else:
        framing_hint = (
            "Standard merchant-scope trigger. "
            "Lead with the exact number from trigger_payload. "
            "Sentence 3: binary YES/NO CTA."
        )

    user_prompt = _USER_TEMPLATE.format(
        facts=json.dumps(facts, ensure_ascii=False, default=str)[:2000],
        prior_bodies=prior_str,
        router_reason=router_reason or "N/A",
        intent_context=intent_context or "N/A",
        mode_instruction=_MODE.get(mode, _MODE["normal"]),
        framing_hint=framing_hint,
    )

    try:
        result: ComposedMessage = await _client.chat.completions.create(
            model=config.COMPOSER_MODEL,
            max_tokens=500,
            temperature=0.0,
            response_model=ComposedMessage,
            max_retries=config.INSTRUCTOR_MAX_RETRIES,
            messages=[
                {"role": "system", "content": _SYSTEM},
                {"role": "user",   "content": user_prompt},
            ],
        )
        return result

    except Exception as e:
        # Log for debugging; return None so caller suppresses gracefully
        print(
            f"[composer] FAILED trigger={trigger_id} conv={conversation_id} "
            f"{type(e).__name__}: {e}"
        )
        return None
